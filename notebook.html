cat_summary = (
    df.groupby('category')
      .agg(
          total_rev=('revenue', 'sum'),
          avg_rev=('revenue', 'mean'),
          transactions=('revenue', 'count'),
      )
      .sort_values('total_rev', ascending=False)
      .reset_index()
)

# Detect anomalies using z-score
from scipy import stats

z_scores = np.abs(stats.zscore(df['revenue'].dropna()))
outliers = df[z_scores > 3.0]

print(f"Outlier transactions: {len(outliers)} ({len(outliers)/len(df)*100:.2f}%)")
print(f"Max outlier value: ${outliers['revenue'].max():,.2f}")
print(f"\nTop 3 outlier products:")
print(outliers.nlargest(3, 'revenue')[['date','product','revenue']].to_string(index=False))
